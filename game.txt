    function speakText(text) {
      if (!soundEnabled || !('speechSynthesis' in window)) return;
      try {
        window.speechSynthesis.cancel();
        const cleanText = text.replace(/<[^>]*>/g, ' ').replace(/[📍🧰💡🗣️🔊👀🎯🔢📏💧⚖️🪓🔪🆘]/g, '');
        const utterance = new SpeechSynthesisUtterance(cleanText);
        utterance.lang = 'id-ID';
        utterance.rate = 0.92;
        window.speechSynthesis.speak(utterance);
      } catch (e) {
        // Abaikan jika browser tidak mendukung TTS
      }
    }

    function playSound(type) {
      if (!soundEnabled || typeof Tone === 'undefined') return;
      try {
        if (Tone.context.state !== 'running') {
          Tone.start();
        }
        const synth = new Tone.Synth().toDestination();
        if (type === 'correct') {
          synth.triggerAttackRelease('C5', '16n', Tone.now());
          synth.triggerAttackRelease('E5', '16n', Tone.now() + 0.1);
          synth.triggerAttackRelease('G5', '8n', Tone.now() + 0.2);
        } else if (type === 'wrong') {
          synth.triggerAttackRelease('E3', '8n', Tone.now());
          synth.triggerAttackRelease('C3', '8n', Tone.now() + 0.12);
        } else if (type === 'click') {
          synth.triggerAttackRelease('A4', '32n', Tone.now());
        } else if (type === 'win') {
          synth.triggerAttackRelease('C5', '8n', Tone.now());
          synth.triggerAttackRelease('E5', '8n', Tone.now() + 0.12);
          synth.triggerAttackRelease('G5', '8n', Tone.now() + 0.24);
          synth.triggerAttackRelease('C6', '4n', Tone.now() + 0.36);
        }
      } catch (e) {
        // Abaikan jika browser memblokir autoplay audio sebelum interaksi
      }
    }

    function toggleSound() {

      const btn = document.getElementById('btn-sound');
      btn.textContent = soundEnabled ? '🔊' : '🔇';
      if (soundEnabled) playSound('click');
    }

    function parseStudentNumber(rawStr) {
      if (rawStr === null || rawStr === undefined) return NaN;
      const cleaned = String(rawStr).trim().replace(/\s+/g, '').replace(',', '.');
      if (cleaned === '') return NaN;
      return Number(cleaned);
    }

    function isNearlyEqual(a, b) {
      if (Number.isNaN(a) || Number.isNaN(b)) return false;
      return Math.abs(a - b) <= 0.02;
    }

    function gradeKnownInputs(prefix, knownFields) {
      let allFilled = true;
      let allCorrect = true;
      const results = [];

      knownFields.forEach(field => {
        const inp = document.getElementById(`${prefix}-inp-${field.key}`);
        const val = parseStudentNumber(inp ? inp.value : '');
        if (Number.isNaN(val)) allFilled = false;
        const ok = isNearlyEqual(val, field.expected);
        if (!ok) allCorrect = false;
        results.push({ field, inp, ok });
      });

      return { allFilled, allCorrect, results };
    }

    function lockKnownInputs(prefix, gradeResult) {
      gradeResult.results.forEach(({ field, inp, ok }) => {
        const wrap = document.getElementById(`${prefix}-wrap-${field.key}`);
        const fb = document.getElementById(`${prefix}-fb-${field.key}`);
        if (inp) inp.disabled = true;
        if (ok) {
          wrap.className = 'bg-emerald-100 border-2 border-emerald-600 rounded-xl p-2 flex flex-col gap-1';
          fb.className = 'text-[10px] font-extrabold text-emerald-800 mt-0.5';
          fb.textContent = `✓ Tepat (${field.key} = ${String(field.expected).replace('.', ',')} ${field.unit})`;
        } else {
          wrap.className = 'bg-rose-100 border-2 border-rose-600 rounded-xl p-2 flex flex-col gap-1';
          fb.className = 'text-[10px] font-extrabold text-rose-800 mt-0.5';
          fb.textContent = `✕ Seharusnya ${field.key} = ${String(field.expected).replace('.', ',')} ${field.unit}`;
        }
      });
    }

    function placePinOnHotspot(hotspotId) {
      if (isAnswered) return;
      playSound('click');

      // Pastikan satu jenis pin hanya terpasang pada satu titik
      Object.keys(placedPins).forEach(key => {
        if (placedPins[key] === activePinType) {
          delete placedPins[key];
        }
      });

      placedPins[hotspotId] = activePinType;
      renderLeverCanvas(MISSIONS[currentMissionIndex]);

      // Pindah otomatis ke pin berikutnya yang belum dipasang
      const used = Object.values(placedPins);
      const nextPin = ['T', 'B', 'K'].find(p => !used.includes(p));
      if (nextPin) selectPinType(nextPin, true);
    }

    function renderLeverCanvas(mission) {
      const canvas = document.getElementById('lever-interactive-canvas');
      const tool = TOOLS_DATA.find(t => t.id === mission.toolId);

      const pinColors = {
        T: 'bg-rose-600 border-white text-white',
        B: 'bg-blue-600 border-white text-white',
        K: 'bg-emerald-600 border-white text-white'
      };
      const pinNames = {
        T: 'T (Tumpu)',
        B: 'B (Beban)',
        K: 'K (Kuasa)'
      };

      const hotspotsHTML = mission.hotspots.map(spot => {
        const assigned = placedPins[spot.id];
        const badgeClass = assigned
          ? `${pinColors[assigned]} scale-105 shadow-lg`
          : 'bg-amber-400 hover:bg-amber-300 border-amber-900 text-amber-950 animate-pulse';
        const text = assigned ? pinNames[assigned] : '?';

        return `
          <button type="button"
            onclick="placePinOnHotspot('${spot.id}')"
            title="${spot.labelHint}"
            style="left: ${spot.x}%; top: ${spot.y}%;"
            class="absolute -translate-x-1/2 -translate-y-1/2 px-2.5 py-1 rounded-full border-2 font-game text-xs font-bold cursor-pointer transition z-10 ${badgeClass}">
            ${text}
          </button>
        `;
      }).join('');

      canvas.innerHTML = `
        <div class="w-40 h-40 sm:w-48 sm:h-48 pointer-events-none select-none">
          ${tool.icon}
        </div>
        ${hotspotsHTML}
      `;
    }

    function checkLeverAnalysis() {
      if (isAnswered) return;
      const mission = MISSIONS[currentMissionIndex];

      // Validasi apakah siswa sudah melengkapi Langkah 1, Langkah 2, dan Langkah 3 (Input Diketahui, Ditanyakan, Rumus & Hasil)
      const placedCount = Object.keys(placedPins).length;
      const knownGrade = gradeKnownInputs('kn-lv', mission.knownFields);
      const studentAnsVal = parseStudentNumber(document.getElementById('ans-lv-inp')?.value);

      if (placedCount < 3 || !selectedLeverType || !knownGrade.allFilled || !selectedAskedLever || !selectedFormulaLever || Number.isNaN(studentAnsVal)) {
        playSound('wrong');
        const feedbackArea = document.getElementById('feedback-area');
        feedbackArea.classList.remove('hidden', 'bg-emerald-100', 'border-emerald-500', 'text-emerald-950', 'bg-rose-100', 'border-rose-500', 'text-rose-950');
        feedbackArea.classList.add('bg-amber-100', 'border-amber-500', 'text-amber-950', 'animate-shake');
        document.getElementById('feedback-icon').textContent = '⚠️';
        document.getElementById('feedback-title').textContent = 'Jawaban Belum Lengkap!';
        document.getElementById('feedback-desc').textContent = 'Lengkapi Langkah 1 (pin T, B, K), Langkah 2 (Jenis Tuas), dan Langkah 3 (ketik semua angka Diketahui, pilih Ditanyakan, pilih Rumus, dan ketik angka Hasil Hitungan).';
        document.getElementById('btn-next').classList.add('hidden');
        return;
      }

      isAnswered = true;
      document.getElementById('btn-next').classList.remove('hidden');

      const allPinsCorrect = mission.hotspots.every(spot => placedPins[spot.id] === spot.expected);
      const leverTypeCorrect = selectedLeverType === mission.correctLeverType;
      const knownCorrect = knownGrade.allCorrect;
      const askedCorrect = selectedAskedLever === mission.correctAsked;
      const formulaCorrect = selectedFormulaLever === mission.correctFormula;
      const answerCorrect = isNearlyEqual(studentAnsVal, mission.answerField.expected);
      const isFullCorrect = allPinsCorrect && leverTypeCorrect && knownCorrect && askedCorrect && formulaCorrect && answerCorrect;

      // Kunci seluruh tombol & input
      const submitBtn = document.getElementById('btn-submit-lever');
      submitBtn.disabled = true;
      submitBtn.classList.add('opacity-50', 'cursor-not-allowed');
      document.querySelectorAll('.lever-type-btn').forEach(btn => btn.disabled = true);

      lockKnownInputs('kn-lv', knownGrade);
      lockAndGradeStepOptions('as-lv', mission.askedOptions, selectedAskedLever, mission.correctAsked);
      lockAndGradeStepOptions('fm-lv', mission.formulaOptions, selectedFormulaLever, mission.correctFormula);
      lockAnswerInput('ans-lv', mission.answerField, answerCorrect);

      // Tampilkan posisi pin yang benar pada gambar alat sebagai evaluasi pasca-jawab
      mission.hotspots.forEach(spot => {
        placedPins[spot.id] = spot.expected;
      });
      renderLeverCanvas(mission);

      const feedbackArea = document.getElementById('feedback-area');
      const feedbackIcon = document.getElementById('feedback-icon');
      const feedbackTitle = document.getElementById('feedback-title');
      const feedbackDesc = document.getElementById('feedback-desc');
      const btnNext = document.getElementById('btn-next');

      feedbackArea.classList.remove('hidden', 'bg-amber-100', 'border-amber-500', 'text-amber-950', 'bg-emerald-100', 'border-emerald-500', 'text-emerald-950', 'bg-rose-100', 'border-rose-500', 'text-rose-950', 'animate-shake');

      const romanTypes = { 1: 'Tuas Jenis I', 2: 'Tuas Jenis II', 3: 'Tuas Jenis III' };
      const correctFormulaLabel = mission.formulaOptions.find(o => o.id === mission.correctFormula).label;

      if (isFullCorrect) {
        score += 10;
        document.getElementById('score-display').textContent = score;
        playSound('correct');

        feedbackArea.classList.add('bg-emerald-100', 'border-emerald-500', 'text-emerald-950');
        feedbackIcon.textContent = '✅';
        feedbackTitle.textContent = `Analisis & Hitungan Sempurna! +10 Poin (${romanTypes[mission.correctLeverType]})`;
        feedbackDesc.textContent = mission.explanation;
      } else {
        playSound('wrong');
        let mistakes = [];
        if (!allPinsCorrect) mistakes.push('letak pin T, B, K');

        if (!knownCorrect) mistakes.push('angka pada Diketahui');
        if (!askedCorrect) mistakes.push('bagian Ditanyakan');
        if (!formulaCorrect) mistakes.push(`rumus (${correctFormulaLabel})`);
        if (!answerCorrect) mistakes.push(`hasil hitungan (${String(mission.answerField.expected).replace('.', ',')} ${mission.answerField.unit})`);

        feedbackArea.classList.add('bg-rose-100', 'border-rose-500', 'text-rose-950');
        feedbackIcon.textContent = '❌';
        feedbackTitle.textContent = `Perlu Dicermati pada: ${mistakes.join(', ')}`;
        feedbackDesc.textContent = mission.explanation;
      }

      btnNext.innerHTML = currentMissionIndex === MISSIONS.length - 1
        ? '<span>Lihat Hasil Akhir</span> <span>🏆</span>'
        : '<span>Misi Berikutnya</span> <span>➡️</span>';

    function showVictoryModal() {
      playSound('win');
      document.getElementById('final-score').textContent = `${score} / ${MISSIONS.length * 10}`;

      let predicate = 'Terus Berlatih Mengenal Pesawat Sederhana!';
      if (score >= 90) {
        predicate = '🌟 Predikat: Master Pesawat Sederhana Lereng Sindoro–Sumbing!';
      } else if (score >= 70) {
        predicate = '👏 Predikat: Asisten Cerdas Petani Temanggung!';
      } else if (score >= 50) {
        predicate = '👍 Predikat: Penjelajah Sains Pemula!';
      }
      document.getElementById('final-predicate').textContent = predicate;

      const modal = document.getElementById('victory-modal');
      modal.classList.remove('hidden');
      modal.classList.add('flex');
    }

    function resetGame() {
      playSound('click');
      closeModal('victory-modal');
      currentMissionIndex = 0;
      score = 0;
      document.getElementById('score-display').textContent = '0';
      loadMission(0);
    }

    // Inisialisasi saat halaman selesai dimuat
    window.onload = function () {
      renderSidebarTools('all');
      loadMission(0);
    };
